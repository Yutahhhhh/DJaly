cmake_minimum_required(VERSION 3.21)
project(djaly_mixxx_host_seam LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Network)
add_executable(djaly-mixxx-host-seam src/main.cpp src/host.cpp src/unavailable_backend.cpp)
target_link_libraries(djaly-mixxx-host-seam PRIVATE Qt6::Core Qt6::Network djaly-junction-core)
target_compile_options(djaly-mixxx-host-seam PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
  add_test(NAME protocol-seam COMMAND "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.test.mjs")
  set_tests_properties(protocol-seam PROPERTIES ENVIRONMENT "DJALY_TEST_HOST=$<TARGET_FILE:djaly-mixxx-host-seam>")
endif()

include(cmake/junction.cmake)
add_executable(junction-core-tests tests/junction/harness.cpp tests/junction/clock_test.cpp tests/junction/media_transport_test.cpp tests/junction/asset_cache_test.cpp tests/junction/program_output_test.cpp tests/junction/producer_tap_test.cpp tests/junction/validation_capture_test.cpp tests/junction/authority_test.cpp)
target_link_libraries(junction-core-tests PRIVATE djaly-junction-core)
add_test(NAME junction-core COMMAND junction-core-tests)
